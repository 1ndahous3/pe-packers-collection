cmake_minimum_required(VERSION 3.24)
project(packman_collection LANGUAGES CXX ASM_MASM)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "Use MSVC Win32")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
set(SRC "${PACKMAN_SOURCE}")
if(NOT EXISTS "${SRC}/Packman/Packer/CPackmanPacker.cpp")
 message(FATAL_ERROR "PACKMAN_SOURCE must point to the extracted Packman.source-1.0 directory")
endif()
file(GLOB CORE "${SRC}/Packman/Packer/*.cpp" "${SRC}/Packman/Packer/Rippers/*.cpp" "${SRC}/Packman/Packer/Loader/*.cpp" "${SRC}/Packman/Packer/Loader/asm/*.asm")
add_library(engine STATIC ${CORE} "${SRC}/Packman/filetools.cpp" "${SRC}/Packman/remem.cpp")
target_include_directories(engine PUBLIC "${SRC}/Packman/Packer" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${SRC}/Packman/Packer/Loader/asm")
# Old SDK macro; the Win32 IMAGE_BASE_RELOCATION structure is still eight bytes.
target_compile_definitions(engine PUBLIC UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS IMAGE_SIZEOF_BASE_RELOCATION=8)
add_executable(packman-cli cli.cpp)
target_link_libraries(packman-cli PRIVATE engine)
add_library(aPLib SHARED "${SRC}/aPLib/main.cpp" "${SRC}/aPLib/unpacker.asm" "${SRC}/aPLib/main.def")
target_link_libraries(aPLib PRIVATE "${SRC}/aPLib/aPLib/aplib.lib")
file(READ "${SRC}/lzma/complib.vcproj" LZMA_PROJECT)
string(REGEX MATCHALL "RelativePath=\"[^\"]+\\.cpp\"" LZMA_MATCHES "${LZMA_PROJECT}")
set(LZMA_CPP)
foreach(ENTRY IN LISTS LZMA_MATCHES)
 string(REGEX REPLACE "^RelativePath=\"|\"$" "" ENTRY "${ENTRY}")
 string(REPLACE "\\" "/" ENTRY "${ENTRY}")
 list(APPEND LZMA_CPP "${SRC}/lzma/${ENTRY}")
endforeach()
add_library(lzma SHARED ${LZMA_CPP} "${SRC}/lzma/unpacker.asm" "${SRC}/lzma/main.def")
target_link_libraries(lzma PRIVATE oleaut32)
# Preserve the original vcproj Release definitions: the ASM decoder uses DWORD probabilities.
target_compile_definitions(lzma PRIVATE COMPRESS_MF_BT COMPRESS_MF_BT4 _LZMA_PROB32)
foreach(TARGET packman-cli aPLib lzma)
 target_compile_definitions(${TARGET} PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
 # The original assembler objects and bundled aPLib library have no SafeSEH metadata.
 target_link_options(${TARGET} PRIVATE /SAFESEH:NO /Brepro /INCREMENTAL:NO)
endforeach()
